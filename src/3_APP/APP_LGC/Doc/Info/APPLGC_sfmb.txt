/*********************************
 * APPLGC_Init
 *********************************/
t_eReturnCode APPLGC_Init(void)
{
    t_eReturnCode Ret_e = RC_OK;
    t_uint8 idxAgent_u8; 
    t_uint8 idxSrv_u8;
    t_uint8 idxAct_u8;
    t_uint8 idxSns_u8;
    
    //----- Set Service Init -----//
    for(idxSrv_u8 = (t_uint8)0 ; (idxSrv_u8 < APPLGC_SRV_NB) && (Ret_e == RC_OK) ; idxSrv_u8++)
    {
        g_srvInfo_as[idxSrv_u8].health_e = APPLGC_SRV_HEALTH_OK;
        g_srvInfo_as[idxSrv_u8].state_e = APPLGC_SRV_STATE_NB;

        Ret_e = SMB_SecureBlockInit(&g_sfmb_srvInfo_as[idxSrv_u8],
                                    &g_srvInfo_as[idxSrv_u8],
                                    (t_uint16)(sizeof(t_sAPPLGC_ServiceInfo)),
                                    (t_uint8)5);
    }

    for(idxAct_u8 = (t_uint8)0 ; (idxAct_u8 < APPACT_ACTUATOR_NB) && (Ret_e == RC_OK) ; idxAct_u8++)
    {
        Ret_e = SMB_SecureBlockInit(&g_sfmb_actGetValues_as[idxAct_u8],
                                    &g_actGetValues_as[idxAct_u8],
                                    (t_uint16)(sizeof(t_uAPPACT_GetValue)),
                                    (t_uint8)5);

        if(Ret_e == RC_OK)
        {
            Ret_e = SMB_SecureBlockInit(&g_sfmb_actSetValues_as[idxAct_u8],
                                        &g_actSetValues_as[idxAct_u8],
                                        (t_uint16)(sizeof(t_uAPPACT_SetValue)),
                                        (t_uint8)5);
        }
    }
    for(idxSns_u8 = (t_uint8)0 ; (idxSns_u8 < APPSNS_SENSOR_NB) && (Ret_e == RC_OK) ; idxSns_u8++)
    {
        Ret_e = SMB_SecureBlockInit(&g_sfmb_snsValues_as[idxSns_u8],
                                    &g_snsValues_af32[idxSns_u8],
                                    (t_uint16)(sizeof(t_float32)),
                                    (t_uint8)5);
    }

    //---- Set Agent Init -----//
    for(idxAgent_u8 = (t_uint8)0 ; (idxAgent_u8 < APPLGC_AGENT_NB) && (Ret_e == RC_OK) ; idxAgent_u8++)
    {
        Ret_e = c_AppLGc_AgentFunc_apf[idxAgent_u8].init_pcb();
    }

    Ret_e = APPSDM_AddCallbackEvnt(s_APPLGC_DiagnosticEvent);

    return Ret_e;
}

/*********************************
 * APPLGC_Cyclic
 *********************************/
t_eReturnCode APPLGC_Cyclic(void)
{
    t_eReturnCode Ret_e = RC_OK;
    // code to run every x milliseconds, config in APPSYS_ConfigPrivate.h

    switch (g_AppLgc_ModState_e)
    {
    case STATE_CYCLIC_CFG:
    {
        Ret_e = s_APPLGC_ConfigurationState();
        if(Ret_e == RC_OK)
        {
            g_AppLgc_ModState_e = STATE_CYCLIC_WAITING;
        }
        break;
    }

    case STATE_CYCLIC_WAITING:
    {
        // nothing to do, just wait all module are Ope
        break;
    }
    case STATE_CYCLIC_PREOPE:
    {
        Ret_e = s_APPLGC_PreOperational();

        if(Ret_e == RC_OK)
        {
            g_AppLgc_ModState_e = STATE_CYCLIC_OPE;
        }
    
        break;
    }
    case STATE_CYCLIC_OPE:
    {
        Ret_e = s_APPLGC_Operational();

        break;
    }
    case STATE_CYCLIC_ERROR:
    {
        break;
    }
    case STATE_CYCLIC_BUSY:
    default:
        Ret_e = RC_OK;
        break;
    }
    return Ret_e;
}

/*********************************
 * APPLGC_SetServiceHealth
 *********************************/
t_eReturnCode APPLGC_SetServiceHealth(t_eAPPLGC_SrvList f_service_e, t_eAPPLGC_SrvHealth f_srvHealth_e)
{
    t_eReturnCode Ret_e = RC_OK;

    if((f_service_e >= APPLGC_SRV_NB)
    || (f_srvHealth_e >= APPLGC_SRV_HEALTH_NB))
    {
        Ret_e = RC_ERROR_PARAM_INVALID;

    {

        g_srvInfo_as[f_service_e].health_e = f_srvHealth_e; 
    }

    return Ret_e;

 *********************************/

{
    t_eReturnCode Ret_e;

    if(f_actuators_e >= APPACT_ACTUATOR_NB)
    {
        Ret_e = RC_ERROR_PARAM_INVALID;

                            f_actValue_pu);
        if(Ret_e != RC_OK)
        {
            ASSERT((t_uint32)Ret_e);
        }
    }

    return Ret_e;
}

/*********************************
 * s_APPLGC_Operational
 *********************************/
static t_eReturnCode s_APPLGC_Operational(void)
{
    t_eReturnCode Ret_e;
    t_uint8 idxAgent_u8;

    if(g_resetSrvState_b == (t_bool)True)
    {
        Ret_e = s_APPLGC_ResetSrvState();
        if(Ret_e == RC_OK)
        {
            g_resetSrvState_b = (t_bool)False;
        }
    }
    if(Ret_e == RC_OK)
    {
        //------ Get Sensors Values for this cyclic -----//
        Ret_e = s_APPLGC_GetSnsValues();
    }
    if(Ret_e == RC_OK)
    {
        Ret_e = s_APPLGC_GetActValues();
    }

    //----- Call Agent Periodic Task Depending on Coordinator -----//
    if(Ret_e == RC_OK)
    {   
        for(idxAgent_u8 = (t_uint8)0 ; (idxAgent_u8 < APPLGC_AGENT_NB) &&  (Ret_e >= RC_OK) ; idxAgent_u8++)
        {
            
            Ret_e = c_AppLGc_AgentFunc_apf[idxAgent_u8].PeriodTask_pcb( (t_float32 *)g_snsValues_af32,
                                                                        (t_sAPPLGC_ServiceInfo *)g_srvInfo_as);
        }
    }

    if(Ret_e >= RC_OK)
    { 
        Ret_e = s_APPLGC_SetActValues();
    }
    return Ret_e;
}

/*********************************
 * s_APPLGC_GetSnsValues
 *********************************/
static t_eReturnCode s_APPLGC_GetSnsValues(void)
{
    t_eReturnCode Ret_e = RC_OK;
    t_sAPPSNS_SnsValueInfo snsInfo_s;
    t_uint8 idxSns_u8 = (t_uint8)0;

    for(idxSns_u8 = (t_uint8)0 ; (idxSns_u8 < APPSNS_SENSOR_NB) && (Ret_e == RC_OK) ; idxSns_u8++)
    {
        //----- Reset Container values -----//
        snsInfo_s.isValueOK_b = (t_bool)False;
        snsInfo_s.rawValue_f32 = (t_float32)0.0;
        snsInfo_s.SnsValue_f32 = (t_float32)0.0;

        Ret_e = APPSNS_Get_SnsValue((t_eAPPSNS_Sensors)idxSns_u8, &snsInfo_s);

        if(Ret_e == RC_OK)
        {
            if(snsInfo_s.isValueOK_b == (t_bool)False)
            {
                snsInfo_s.SnsValue_f32 = (t_float32)0.0f;
            }

            SMB_Write(&g_sfmb_snsValues_as[idxSns_u8], &snsInfo_s.SnsValue_f32);
            if(Ret_e != RC_OK)
            {
                ASSERT((t_uint32)Ret_e);
            }
        }
    }
    
    return Ret_e;
}

/*********************************
 * s_APPLGC_GetActValues
 *********************************/
static t_eReturnCode s_APPLGC_GetActValues(void)
{
    t_eReturnCode Ret_e = RC_OK;
    t_uAPPACT_GetValue actGetValue_u;
    t_uint8 idxAct_u8 = (t_uint8)0;

    for(idxAct_u8 = (t_uint8)0 ; (idxAct_u8 < APPSNS_SENSOR_NB) && (Ret_e == RC_OK) ; idxAct_u8++)
    {
        (void)SafeMem_memset(&actGetValue_u, (t_uint16)0, sizeof(t_uAPPACT_GetValue));

        Ret_e = APPACT_Get_ActValue(idxAct_u8, &actGetValue_u);

        if(Ret_e == RC_OK)
        {
            Ret_e = SMB_Write(&g_sfmb_actGetValues_as[idxAct_u8], &actGetValue_u);

            if(Ret_e != RC_OK)
            {
                ASSERT((t_uint16)Ret_e);
            }
        }
    }
    
    return Ret_e;
}
/*********************************
 * s_APPLGC_SetActValues
 *********************************/
static t_eReturnCode s_APPLGC_SetActValues(void)
{
    t_eReturnCode Ret_e = RC_OK;
    t_uint8 idxAct_u8 = (t_uint8)0;
    t_eAPPACT_Actuators actuatorLabel_e;
    t_uAPPACT_SetValue actSetValue_u;


    //----- Loop on every Actuators For this Service -----//
    for(idxAct_u8 = (t_uint8)0 ; (idxAct_u8 < APPACT_ACTUATOR_NB) && (Ret_e == RC_OK) ; idxAct_u8++)
    {
        Ret_e = SMB_Read(&g_sfmb_actSetValues_as[idxAct_u8], &actSetValue_u);

        if(Ret_e == RC_OK)
        {
            Ret_e = APPACT_Set_ActValue(idxAct_u8, (t_uAPPACT_SetValue)actSetValue_u);
        }
    }

    return Ret_e;
}

/*********************************
 * s_APPLGC_ResetSrvState
 *********************************/
static t_eReturnCode s_APPLGC_ResetSrvState(void)
{
    t_eReturnCode Ret_e = RC_OK;
    t_uint8 idxSrv_u8;

    Ret_e = APPSDM_ResetDiagEvnt();

    if(Ret_e == RC_OK)
    {
        for(idxSrv_u8 = (t_uint8)0 ; 
            (idxSrv_u8 < APPLGC_SRV_NB)
        &&  (Ret_e == RC_OK) ; 
        idxSrv_u8++)
        {
            Ret_e = APPLGC_SetServiceHealth((t_eAPPLGC_SrvList)idxSrv_u8, 
                                            APPLGC_SRV_HEALTH_OK);
        }
    }

    return Ret_e;
}
